import { getAuth, RecaptchaVerifier, signInWithPhoneNumber } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-auth.js"; const auth = getAuth(app); const sendCodeButton = document.getElementById("sendCode"); const verifyCodeButton = document.getElementById("verifyCode"); const phoneInput = document.getElementById("phoneNumber"); const codeInput = document.getElementById("verificationCode"); const verificationSection = document.getElementById("verificationSection"); const privateContent = document.getElementById("privateContent"); const appVerifier = new RecaptchaVerifier(auth, "recaptcha-container", { size: "normal" }); sendCodeButton.addEventListener("click", async () => { const phoneNumber = phoneInput.value.trim(); if (!phoneNumber) { alert("اكتب رقم الهاتف أولاً"); return; } try { const confirmationResult = await signInWithPhoneNumber(auth, phoneNumber, appVerifier); window.confirmationResult = confirmationResult; verificationSection.style.display = "block"; alert("تم إرسال رمز التحقق"); } catch (error) { console.error(error); alert("حدث خطأ أثناء إرسال الرمز"); } }); verifyCodeButton.addEventListener("click", async () => { const code = codeInput.value.trim(); if (!code) { alert("اكتب رمز التحقق"); return; } try { await window.confirmationResult.confirm(code); privateContent.style.display = "block"; alert("تم تسجيل الدخول بنجاح ✅"); } catch (error) { console.error(error); alert("رمز التحقق غير صحيح"); } });
